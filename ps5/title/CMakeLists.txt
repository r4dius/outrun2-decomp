cmake_minimum_required(VERSION 3.20)
project(OutRunPS5 LANGUAGES CXX)
option(OR2_PS5_HOST_TEST "Compile the same application against desktop SDL for integration checks" OFF)
option(OR2_PS5_OPENGL "Use the native OpenGL GPU renderer" OFF)
option(OR2_PS5_VULKAN "Use the native Vulkan RADV GPU renderer" OFF)
option(OR2_PS5_NATIVE_ARCHIVE "Build an application archive for native title packaging" OFF)
if(OR2_PS5_OPENGL AND OR2_PS5_VULKAN)
    message(FATAL_ERROR "Select one PS5 renderer per build: OpenGL or Vulkan")
endif()
get_filename_component(OR2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT OR2_PS5_HOST_TEST AND NOT PS5)
    message(FATAL_ERROR "Use the ps5-payload-sdk prospero.cmake toolchain or OR2_PS5_HOST_TEST=ON")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(OR2_PS5_HOST_TEST)
    find_package(SDL2 REQUIRED)
    find_package(ZLIB REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(OR2_MEDIA REQUIRED IMPORTED_TARGET libavformat libavcodec libswscale libswresample libavutil vorbisfile)
else()
    set(OR2_PS5_MEDIA_ROOT "${PS5_PAYLOAD_SDK}/target/user/homebrew" CACHE PATH "PacBrew SDL/media installation")
    set(OR2_HBROOT "${OR2_PS5_MEDIA_ROOT}")
    find_package(SDL2 CONFIG REQUIRED PATHS "${OR2_HBROOT}/lib/cmake/SDL2" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
endif()
file(GLOB OR2_ENGINE_SOURCES CONFIGURE_DEPENDS "${OR2_ROOT}/src/driving/*.cpp" "${OR2_ROOT}/src/platform/*.cpp" "${OR2_ROOT}/src/system/*.cpp" "${OR2_ROOT}/src/runtime/*.cpp" "${OR2_ROOT}/src/enhancements/*.cpp")
add_library(outrun_ps5_engine STATIC ${OR2_ENGINE_SOURCES} "${OR2_ROOT}/src/asset_views.cpp" "${OR2_ROOT}/src/pc_pmt.cpp")
target_sources(outrun_ps5_engine PRIVATE "${OR2_ROOT}/src/ports/pecompact_memory.cpp")
target_include_directories(outrun_ps5_engine PUBLIC "${OR2_ROOT}/src")
target_compile_definitions(outrun_ps5_engine PUBLIC OR2_X87_FAST=1 OR2_HAVE_VORBIS=1 OUTRUN_WITH_FFMPEG=1)
target_compile_options(outrun_ps5_engine PRIVATE -O3 -ffp-contract=off -fno-fast-math -Wall -Wextra)
set(OR2_PS5_SOURCES
    ../source/main.cpp "${OR2_ROOT}/src/input/dual_sense.cpp" "${OR2_ROOT}/src/input/runtime_adapter.cpp"
    ../source/platform_api.cpp ../source/renderer.cpp ../source/sdl_backend.cpp ../source/audio.cpp ../source/setup_display.cpp)
if(OR2_PS5_OPENGL)
    list(APPEND OR2_PS5_SOURCES ../source/gl_api.cpp ../source/gl_display.cpp ../source/gl_d3d9.cpp)
endif()
set(OR2_PS5_VK_SOURCES ../source/vk_api.cpp ../source/vk_context.cpp ../source/vk_display.cpp ../source/vk_d3d9.cpp)
# Vulkan's sType-first aggregate initialization zeroes every remaining field.
set_source_files_properties(${OR2_PS5_VK_SOURCES} PROPERTIES COMPILE_OPTIONS "-Wno-missing-field-initializers")
if(OR2_PS5_VULKAN)
    list(APPEND OR2_PS5_SOURCES ${OR2_PS5_VK_SOURCES})
endif()
if(OR2_PS5_NATIVE_ARCHIVE)
    add_library(OutRun2006-PS5 STATIC ${OR2_PS5_SOURCES})
    target_compile_definitions(OutRun2006-PS5 PRIVATE OR2_PS5_NATIVE_TITLE=1)
    set_source_files_properties(../source/main.cpp PROPERTIES COMPILE_DEFINITIONS "main=or2_ps5_main")
else()
    add_executable(OutRun2006-PS5 ${OR2_PS5_SOURCES})
endif()
if(OR2_PS5_OPENGL)
    find_program(OR2_PYTHON python3 REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    set(OR2_GL_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/pc_gl_shaders.hpp")
    add_custom_command(OUTPUT "${OR2_GL_HEADER}" COMMAND "${OR2_PYTHON}" "${OR2_ROOT}/tools/embed_ps5_shaders.py" "${OR2_GL_HEADER}"
        DEPENDS "${OR2_ROOT}/tools/embed_ps5_shaders.py" "${OR2_ROOT}/ps5/shaders/pc_ffp_fsh.glsl" "${OR2_ROOT}/ps5/shaders/pc_ps11_uber_fsh.glsl" "${OR2_ROOT}/ps5/shaders/pc_ps14_uber_fsh.glsl")
    target_sources(OutRun2006-PS5 PRIVATE "${OR2_GL_HEADER}")
    target_include_directories(OutRun2006-PS5 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(OutRun2006-PS5 PRIVATE OR2_PS5_OPENGL=1)
    if(OR2_PS5_HOST_TEST)
        find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
        target_link_libraries(OutRun2006-PS5 PRIVATE OpenGL::OpenGL OpenGL::EGL)
    else()
        if(NOT OR2_PS5_NATIVE_ARCHIVE)
            message(FATAL_ERROR "The PS5 OpenGL backend requires native title packaging, not an ELF loader")
        endif()
        find_package(PS5OpenGL CONFIG REQUIRED NO_CMAKE_FIND_ROOT_PATH)
        target_link_libraries(OutRun2006-PS5 PRIVATE PS5OpenGL::OpenGL)
    endif()
endif()
target_include_directories(OutRun2006-PS5 PRIVATE ../source)
if(OR2_PS5_VULKAN)
    find_program(OR2_PYTHON python3 REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    find_program(OR2_GLSLANG glslangValidator REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    find_program(OR2_SPIRV_VAL spirv-val REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    find_program(OR2_SPIRV_OPT spirv-opt REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    set(OR2_PS5_VK_PROFILE_CACHES "" CACHE STRING "Optional private runtime cache files for offline program specialization")
    option(OR2_PS5_MENU_TRACE "Record bounded menu ownership and geometry diagnostics" OFF)
    if(OR2_PS5_MENU_TRACE)
        target_compile_definitions(OutRun2006-PS5 PRIVATE OR2_PS5_MENU_TRACE=1)
    endif()
    set(OR2_VK_PROFILE_ARGS)
    foreach(profile IN LISTS OR2_PS5_VK_PROFILE_CACHES)
        list(APPEND OR2_VK_PROFILE_ARGS --profile-cache "${profile}")
    endforeach()
    set(OR2_VK_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/pc_vk_shaders.hpp")
    add_custom_command(OUTPUT "${OR2_VK_HEADER}"
        COMMAND "${OR2_PYTHON}" "${OR2_ROOT}/tools/compile_ps5_vulkan_shaders.py" "${OR2_VK_HEADER}"
            --compiler "${OR2_GLSLANG}" --validator "${OR2_SPIRV_VAL}" --optimizer "${OR2_SPIRV_OPT}" ${OR2_VK_PROFILE_ARGS}
        DEPENDS "${OR2_ROOT}/tools/compile_ps5_vulkan_shaders.py" "${OR2_ROOT}/ps5/source/gl_d3d9.cpp"
            "${OR2_ROOT}/switch/shaders/pc_vs_uber_vsh.glsl"
            "${OR2_ROOT}/ps5/shaders/pc_ffp_fsh.glsl" "${OR2_ROOT}/ps5/shaders/pc_ps11_uber_fsh.glsl" "${OR2_ROOT}/ps5/shaders/pc_ps14_uber_fsh.glsl" ${OR2_PS5_VK_PROFILE_CACHES})
    target_sources(OutRun2006-PS5 PRIVATE "${OR2_VK_HEADER}")
    target_compile_definitions(OutRun2006-PS5 PRIVATE OR2_PS5_VULKAN=1)
    target_include_directories(OutRun2006-PS5 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(OR2_PS5_HOST_TEST)
        find_package(Vulkan REQUIRED)
        target_link_libraries(OutRun2006-PS5 PRIVATE Vulkan::Vulkan)
    else()
        if(NOT OR2_PS5_NATIVE_ARCHIVE)
            message(FATAL_ERROR "The Vulkan RADV renderer requires native title packaging; use tools/build_ps5_vulkan.sh")
        endif()
        set(OR2_PS5_RADV_ROOT "" CACHE PATH "Pinned PS5 RADV installation (headers and archive)")
        if(NOT EXISTS "${OR2_PS5_RADV_ROOT}/include/vulkan/vulkan.h" OR NOT EXISTS "${OR2_PS5_RADV_ROOT}/lib/libvulkan_radeon.ps5.a")
            message(FATAL_ERROR "Set OR2_PS5_RADV_ROOT after tools/setup_ps5_vulkan.sh")
        endif()
        target_include_directories(OutRun2006-PS5 PRIVATE "${OR2_PS5_RADV_ROOT}/include")
    endif()
endif()
target_compile_options(OutRun2006-PS5 PRIVATE -O3 -ffp-contract=off -fno-fast-math -Wall -Wextra)
target_link_libraries(OutRun2006-PS5 PRIVATE outrun_ps5_engine SDL2::SDL2 Threads::Threads)
if(OR2_PS5_HOST_TEST)
    target_compile_definitions(OutRun2006-PS5 PRIVATE OR2_PS5_HOST_TEST=1)
    target_link_libraries(outrun_ps5_engine PUBLIC ZLIB::ZLIB PkgConfig::OR2_MEDIA)
    enable_testing()
    enable_language(C)
    add_executable(test_ps5_native_memory ../tests/test_native_memory.c ../source/native_memory.c)
    target_link_libraries(test_ps5_native_memory PRIVATE Threads::Threads)
    add_test(NAME ps5_native_anonymous_memory COMMAND test_ps5_native_memory)
    add_executable(test_ps5_sdl ../tests/test_sdl.cpp ../source/audio.cpp ../source/sdl_backend.cpp)
    target_include_directories(test_ps5_sdl PRIVATE ../source)
    target_link_libraries(test_ps5_sdl PRIVATE outrun_ps5_engine SDL2::SDL2)
    add_test(NAME ps5_sdl_boundaries COMMAND test_ps5_sdl)
    set_tests_properties(ps5_sdl_boundaries PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
    if(OR2_PS5_OPENGL)
        add_executable(test_ps5_gl ../tests/test_gl.cpp ../source/gl_api.cpp ../source/gl_display.cpp ../source/gl_d3d9.cpp "${OR2_GL_HEADER}")
        target_include_directories(test_ps5_gl PRIVATE ../source "${CMAKE_CURRENT_BINARY_DIR}/generated")
        target_link_libraries(test_ps5_gl PRIVATE outrun_ps5_engine OpenGL::OpenGL OpenGL::EGL)
        add_test(NAME ps5_gl_pixels_and_resources COMMAND test_ps5_gl)
        set_tests_properties(ps5_gl_pixels_and_resources PROPERTIES ENVIRONMENT "EGL_PLATFORM=surfaceless")
    endif()
    if(OR2_PS5_VULKAN)
        add_executable(test_ps5_vulkan ../tests/test_vulkan.cpp ${OR2_PS5_VK_SOURCES} "${OR2_VK_HEADER}")
        target_compile_definitions(test_ps5_vulkan PRIVATE OR2_PS5_HOST_TEST=1)
        target_include_directories(test_ps5_vulkan PRIVATE ../source "${CMAKE_CURRENT_BINARY_DIR}/generated")
        target_link_libraries(test_ps5_vulkan PRIVATE outrun_ps5_engine Vulkan::Vulkan)
        add_test(NAME ps5_vulkan_pixels_and_resources COMMAND test_ps5_vulkan)
    endif()
else()
    target_compile_definitions(OutRun2006-PS5 PRIVATE OR2_PS5_PAYLOAD=1)
    target_include_directories(outrun_ps5_engine PUBLIC "${OR2_HBROOT}/include")
    target_link_directories(outrun_ps5_engine PUBLIC "${OR2_HBROOT}/lib")
    # PacBrew static FFmpeg dependencies: group handles archive cycles.
    target_link_libraries(outrun_ps5_engine PUBLIC
        -Wl,--start-group avformat avcodec swscale swresample avutil
        vorbisfile vorbis ogg z bz2 lzma iconv x264 ssl crypto samplerate
        -Wl,--end-group m)
    if(NOT OR2_PS5_NATIVE_ARCHIVE)
        set_target_properties(OutRun2006-PS5 PROPERTIES SUFFIX ".elf")
    endif()
endif()
